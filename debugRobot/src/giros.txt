// Dejamos las originales para no romper dependencias de otros archivos
sensado sensadoActual = {0,0,0};
ESTADOS estadoActual = ESPERANDO_INICIO;
VELOCIDAD velocidadActual = {0,0};

int pasoTest = 0; // Controla la secuencia de la prueba

//==============================================================
//                     VOID SETUP
//==============================================================

void setup (){
  pinMode(BOTON, INPUT);
  inicializarLogger();
  inicializacionSensoresDist(); 
  inicializarMotores(); 
  inicializarEncoders();
  Serial.println("--- INICIO TEST DE ENCODERS ---");
  Serial.println("Presiona el boton para iniciar: Giro Derecha");
}

// Función auxiliar para esperar el botón limpiamente
void esperarBoton() {
    movimiento(FRENO_F, {0,0});
    // Esperar a que se presione
    while(digitalRead(BOTON) == HIGH) { delay(10); }
    // Esperar a que se suelte
    while(digitalRead(BOTON) == LOW) { delay(10); }
    delay(500); // Tiempo de seguridad
}

//==============================================================
//                       VOID LOOP
//==============================================================

void loop(){
    switch(pasoTest) {
        case 0: // Esperar inicio
            esperarBoton();
            Serial.println("-> GIRO DERECHA (Tunea PULSOS_90_GRADOS en config.h si gira de mas o de menos)");
            resetearEncoders();
            pasoTest = 1;
            break;
            
        case 1: // Girando Derecha
            if (abs(verPulsosEncoderA()) < 200) {
                movimiento(GIRAR_DER, {VEL_GIRO_IZQ, VEL_GIRO_DER});
            } else {
                movimiento(FRENO_F, {0,0});
                Serial.println("-> PRESIONA EL BOTON PARA GIRAR A LA IZQUIERDA");
                pasoTest = 2;
            }
            break;
            
        case 2: // Esperar
            esperarBoton();
            Serial.println("-> GIRO IZQUIERDA (Tunea PULSOS_90_GRADOS)");
            resetearEncoders();
            pasoTest = 3;
            break;

        case 3: // Girando Izquierda
            if (abs(verPulsosEncoderA()) < PULSOS_90_GRADOS) {
                movimiento(GIRAR_IZQ, {VEL_GIRO_IZQ, VEL_GIRO_DER});
            } else {
                movimiento(FRENO_F, {0,0});
                Serial.println("-> PRESIONA EL BOTON PARA AVANZAR");
                pasoTest = 4;
            }
            break;
            
        case 4: // Esperar
            esperarBoton();
            Serial.println("-> AVANZANDO (Tunea PULSOS_AVANZAR_BLOQUEANTE)");
            resetearEncoders();
            pasoTest = 5;
            break;

        case 5: // Avanzando
            if (abs(verPulsosEncoderA()) < PULSOS_AVANZAR_BLOQUEANTE) {
                movimiento(AVANZAR, {80, 80}); 
            } else {
                movimiento(FRENO_F, {0,0});
                Serial.println("--- FIN DEL TEST ---");
                pasoTest = 6;
            }
            break;

        case 6: // Fin
            delay(100);
            pasoTest = 0;
            break;
    }
}
